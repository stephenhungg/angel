src/components/ProjectCard.tsx  (excerpt)
==============================================================================
type Project = {
  slug: string;
  name: string;
  blurb: string;
  href: string;
  year: number;
  tags: string[];
};

export function ProjectCard({ p }: { p: Project }) {
  return (
    <a href={p.href} className="group rounded-2xl border ...">
      <h3 className="text-lg">{p.name}</h3>
      <p className="text-sm opacity-70">{p.blurb}</p>
      <ul className="mt-2 flex gap-1">{p.tags.map((t) => <li key={t}>#{t}</li>)}</ul>
    </a>
  );
}
